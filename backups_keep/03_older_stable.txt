class Chemical {
  final String id;
  final String name;
  final String formula;
  final String state;
  final String category;
  final Color color;
  const Chemical({
    required this.id,
    required this.name,
    required this.formula,
    required this.state,
    required this.category,
    required this.color,
  });
}

class ReactionInfo {
  final String name;
  final String equation;
  final String type;
  final String description;
  final List<String> products;
  final bool bubbles;
  final bool precipitate;
  final Color? colorChange;
  const ReactionInfo({
    required this.name,
    required this.equation,
    required this.type,
    required this.description,
    required this.products,
    this.bubbles = false,
    this.precipitate = false,
    this.colorChange,
  });
}

// ─── Effects ───
enum LabEffect { heat, cool, stir, thermometer, dropper }

extension LabEffectX on LabEffect {
  String get label =>
      const ['Heat', 'Cool', 'Stir', 'Thermo', 'Dropper'][index];
  IconData get icon => const [
        Icons.local_fire_department,
        Icons.ac_unit,
        Icons.rotate_right,
        Icons.thermostat,
        Icons.water_drop,
      ][index];
  Color get color => const [
        P.orange,
        P.blueBright,
        P.cyan,
        P.red,
        P.purple,
      ][index];
}

// ─── Equipment ───
enum EquipShape { beaker, flask, testTube, cylinder }

extension EquipShapeX on EquipShape {
  String get label =>
      const ['Beaker', 'Flask', 'Test Tube', 'Cylinder'][index];
  IconData get icon => const [
        Icons.coffee,
        Icons.science,
        Icons.biotech,
        Icons.straighten,
      ][index];
  double get maxVol => const [250.0, 250.0, 50.0, 100.0][index];
  Size get size => const [
        Size(140, 180),
        Size(160, 220),
        Size(60, 210),
        Size(80, 230),
      ][index];
}

class EquipItem {
  final String id;
  final EquipShape shape;
  Offset position;
  final List<String> chemicals = [];
  final Set<LabEffect> effects = {};
  double temperature = 25.0;

  EquipItem({
    required this.id,
    required this.shape,
    required this.position,
  });

  double get totalVolume => chemicals.length * 25.0;
  double get fillRatio => (totalVolume / shape.maxVol).clamp(0.0, 1.0);
  bool get isHot => effects.contains(LabEffect.heat);
  bool get isCold => effects.contains(LabEffect.cool);
  bool get isStirring => effects.contains(LabEffect.stir);
  bool get hasThermo => effects.contains(LabEffect.thermometer);
  bool get hasDropper => effects.contains(LabEffect.dropper);

  void clear() {
    chemicals.clear();
    effects.clear();
    temperature = 25.0;
  }
}

class LabScreen extends StatefulWidget {
  const LabScreen({super.key});
  @override
  State<LabScreen> createState() => _LabScreenState();
}

class _LabScreenState extends State<LabScreen> with TickerProviderStateMixin {
  // UI state
  String category = 'All';
  int navIndex = 1;
  bool showInventory = false;
  bool showToolbox = false;
  String experimentName = 'Neutralization Reaction';

  // Bench items
  final List<EquipItem> equipment = [
    EquipItem(id: 'eq1', shape: EquipShape.beaker, position: const Offset(160, 140)),
    EquipItem(id: 'eq2', shape: EquipShape.flask, position: const Offset(400, 130)),
    EquipItem(id: 'eq3', shape: EquipShape.testTube, position: const Offset(620, 120)),
  ];

  // Drag state
  Chemical? draggingChemical;
  LabEffect? draggingEffect;
  Offset dragPos = Offset.zero;
  OverlayEntry? dragOverlay;
  String? hoveredEquipId;
  String? activeEquipId;

  // Results
  ReactionInfo? currentReaction;

  // Physics
  Timer? _tick;
  double _time = 0;

  // Bench key
  final GlobalKey _benchKey = GlobalKey();

  @override
  void initState() {
    super.initState();
    _tick = Timer.periodic(const Duration(milliseconds: 16), (_) {
      if (!mounted) return;
      _time += 0.016;

      // Update temperature physics
      bool needs = false;
      for (final eq in equipment) {
        if (eq.isHot) {
          final newT = (eq.temperature + 0.5).clamp(25.0, 150.0);
          if ((newT - eq.temperature).abs() > 0.01) {
            eq.temperature = newT;
            needs = true;
          }
        } else if (eq.isCold) {
          final newT = (eq.temperature - 0.8).clamp(0.0, 150.0);
          if ((newT - eq.temperature).abs() > 0.01) {
            eq.temperature = newT;
            needs = true;
          }
        } else if (eq.temperature > 25.0) {
          eq.temperature = (eq.temperature - 0.2).clamp(25.0, 150.0);
          needs = true;
        } else if (eq.temperature < 25.0) {
          eq.temperature = (eq.temperature + 0.2).clamp(0.0, 25.0);
          needs = true;
        }
      }
      if (needs) setState(() {});
    });
  }

  @override
  void dispose() {
    _tick?.cancel();
    dragOverlay?.remove();
    super.dispose();
  }

  // ═══════════════════════════════════════════════════
  // HELPERS
  // ═══════════════════════════════════════════════════
  List<Chemical> get filteredChemicals {
    if (category == 'All') return allChemicals;
    return allChemicals.where((c) => c.category == category).toList();
  }

  void _recomputeReaction() {
    currentReaction = null;
    for (final eq in equipment) {
      if (eq.chemicals.length >= 2) {
        final sorted = [...eq.chemicals]..sort();
        final key = sorted.join('+');
        final rxn = reactions[key];
        if (rxn != null) {
          currentReaction = rxn;
          return;
        }
      }
    }
  }

  // ═══════════════════════════════════════════════════
  // ACTIONS
  // ═══════════════════════════════════════════════════
  void addEquipment(EquipShape shape) {
    final x = 100.0 + (equipment.length * 140);
    setState(() {
      equipment.add(EquipItem(
        id: 'eq_${DateTime.now().microsecondsSinceEpoch}',
        shape: shape,
        position: Offset(x, 130),
      ));
    });
  }

  void removeEquipment(String id) {
    setState(() {
      equipment.removeWhere((e) => e.id == id);
      if (activeEquipId == id) activeEquipId = null;
      _recomputeReaction();
    });
  }

  void reset() {
    setState(() {
      for (final eq in equipment) {
        eq.clear();
      }
      currentReaction = null;
      activeEquipId = null;
    });
  }

  // ═══════════════════════════════════════════════════
  // CHEMICAL DRAG
  // ═══════════════════════════════════════════════════
  void startChemicalDrag(Chemical c, Offset pos) {
    draggingChemical = c;
    dragPos = pos;
    dragOverlay?.remove();
    dragOverlay = OverlayEntry(
      builder: (_) => Positioned(
        left: dragPos.dx - 40,
        top: dragPos.dy - 55,
        child: IgnorePointer(
          child: Material(
            color: Colors.transparent,
            child: _dragBottle(c),
          ),
        ),
      ),
    );
    Overlay.of(context).insert(dragOverlay!);
    setState(() {});
  }

  void updateChemicalDrag(Offset pos) {
    if (dragOverlay == null) return;
    dragPos = pos;
    dragOverlay!.markNeedsBuild();
    final hit = _equipUnderPoint(pos);
    if (hit != hoveredEquipId) setState(() => hoveredEquipId = hit);
  }

  void endChemicalDrag() {
    dragOverlay?.remove();
    dragOverlay = null;
    final hit = _equipUnderPoint(dragPos);
    if (hit != null && draggingChemical != null) {
      _pourInto(hit, draggingChemical!);
    }
    setState(() {
      draggingChemical = null;
      hoveredEquipId = null;
    });
  }

  // ═══════════════════════════════════════════════════
  // EFFECT DRAG
  // ═══════════════════════════════════════════════════
  void startEffectDrag(LabEffect effect, Offset pos) {
    draggingEffect = effect;
    dragPos = pos;
    dragOverlay?.remove();
    dragOverlay = OverlayEntry(
      builder: (_) => Positioned(
        left: dragPos.dx - 60,
        top: dragPos.dy - 25,
        child: IgnorePointer(
          child: Material(
            color: Colors.transparent,
            child: _dragEffectChip(effect),
          ),
        ),
      ),
    );
    Overlay.of(context).insert(dragOverlay!);
    setState(() {});
  }

  void updateEffectDrag(Offset pos) {
    if (dragOverlay == null) return;
    dragPos = pos;
    dragOverlay!.markNeedsBuild();
    final hit = _equipUnderPoint(pos);
    if (hit != hoveredEquipId) setState(() => hoveredEquipId = hit);
  }

  void endEffectDrag() {
    dragOverlay?.remove();
    dragOverlay = null;
    final hit = _equipUnderPoint(dragPos);
    if (hit != null && draggingEffect != null) {
      _applyEffect(hit, draggingEffect!);
    }
    setState(() {
      draggingEffect = null;
      hoveredEquipId = null;
    });
  }

  // ═══════════════════════════════════════════════════
  // HIT TESTING
  // ═══════════════════════════════════════════════════
  String? _equipUnderPoint(Offset globalPos) {
    final benchBox = _benchKey.currentContext?.findRenderObject() as RenderBox?;
    if (benchBox == null) return null;
    final local = benchBox.globalToLocal(globalPos);

    for (final eq in equipment) {
      final s = eq.shape.size;
      final rect = Rect.fromLTWH(
        eq.position.dx - 20,
        eq.position.dy - 20,
        s.width + 40,
        s.height + 40,
      );
      if (rect.contains(local)) return eq.id;
    }
    return null;
  }

  // ═══════════════════════════════════════════════════
  // POUR / APPLY
  // ═══════════════════════════════════════════════════
  void _pourInto(String equipId, Chemical c) {
    setState(() {
      final eq = equipment.firstWhere((e) => e.id == equipId);
      if (!eq.chemicals.contains(c.id)) eq.chemicals.add(c.id);
      activeEquipId = equipId;
      _recomputeReaction();
    });
  }

  void _applyEffect(String equipId, LabEffect effect) {
    setState(() {
      final eq = equipment.firstWhere((e) => e.id == equipId);
      if (eq.effects.contains(effect)) {
        eq.effects.remove(effect);
      } else {
        eq.effects.add(effect);
        if (effect == LabEffect.heat) eq.effects.remove(LabEffect.cool);
        if (effect == LabEffect.cool) eq.effects.remove(LabEffect.heat);
      }
      activeEquipId = equipId;
    });
  }

  // ═══════════════════════════════════════════════════
  // DRAG PREVIEW WIDGETS
  // ═══════════════════════════════════════════════════
  Widget _dragBottle(Chemical c) {
    return SizedBox(
      width: 80, height: 110,

        angle: -0.12,

          decoration: BoxDecoration(
            gradient: const LinearGradient(

              BoxShadow(
                color: P.cyan.withValues(alpha: 0.5),
                blurRadius: 24,
                offset: const Offset(0, 8),

            mainAxisAlignment: MainAxisAlignment.center,

                  color: c.color,

        ),

    );
  }

            color: effect.color.withValues(alpha: 0.6),

        mainAxisSize: MainAxisSize.min,

            effect.label,
            style: const TextStyle(
              color: Colors.white,
              fontSize: 12,
              fontWeight: FontWeight.bold,
            ),
          ),
        ],
      ),
    );
  }

  // ═══════════════════════════════════════════════════
  // EQUIPMENT WIDGET
  // ═══════════════════════════════════════════════════
  Widget _equipmentWidget(EquipItem eq) {
    final isHovered = hoveredEquipId == eq.id;
    final isActive = activeEquipId == eq.id;
    final s = eq.shape.size;

    Color liquidColor = Colors.transparent;
    if (eq.chemicals.isNotEmpty) {
      liquidColor = allChemicals.firstWhere((c) => c.id == eq.chemicals.last).color;
    }
    if (isActive && currentReaction?.colorChange != null) {
      liquidColor = currentReaction!.colorChange!;
    }

    return Stack(
      clipBehavior: Clip.none,
      children: [
        Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            // Badge row
            if (eq.effects.isNotEmpty || eq.temperature != 25.0)
              Container(
                margin: const EdgeInsets.only(bottom: 4),
                padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 3),
                decoration: BoxDecoration(
                  color: P.bg.withValues(alpha: 0.92),
                  borderRadius: BorderRadius.circular(10),
                  border: Border.all(color: P.border),
                ),
                child: Row(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    if (eq.isHot) const Padding(padding: EdgeInsets.only(right: 3), child: Icon(Icons.local_fire_department, size: 10, color: P.orange)),
                    if (eq.isCold) const Padding(padding: EdgeInsets.only(right: 3), child: Icon(Icons.ac_unit, size: 10, color: P.blueBright)),
                    if (eq.isStirring) const Padding(padding: EdgeInsets.only(right: 3), child: Icon(Icons.rotate_right, size: 10, color: P.cyan)),
                    if (eq.hasThermo) const Padding(padding: EdgeInsets.only(right: 3), child: Icon(Icons.thermostat, size: 10, color: P.red)),
                    if (eq.hasDropper) const Padding(padding: EdgeInsets.only(right: 3), child: Icon(Icons.water_drop, size: 10, color: P.purple)),
                    if (eq.effects.isNotEmpty) const SizedBox(width: 3),
                    Text('${eq.temperature.toInt()}°C',
                        style: const TextStyle(
                            color: P.amber, fontSize: 9, fontWeight: FontWeight.bold)),
                  ],
                ),
              ),
            // Equipment with effect overlays (behind + inside)
            SizedBox(
              width: s.width + 80,
              height: s.height + 100,
              child: Stack(
                alignment: Alignment.bottomCenter,
                children: [
                  // Flame UNDER (behind)
                  if (eq.isHot)
                    Positioned(
                      bottom: 0,
                      child: SizedBox(
                        width: s.width + 40,
                        height: 90,
                        child: CustomPaint(painter: FlamePainter(progress: _time)),
                      ),
                    ),
                  // Ice AROUND (behind glass)
                  if (eq.isCold)
                    Positioned(
                      bottom: 20,
                      child: SizedBox(
                        width: s.width + 60,
                        height: s.height + 30,
                        child: CustomPaint(painter: IcePainter(progress: _time)),
                      ),
                    ),
                  // Steam ABOVE
                  if (eq.isHot && eq.chemicals.isNotEmpty)
                    Positioned(
                      top: 0,
                      child: SizedBox(
                        width: s.width + 40,
                        height: 90,
                        child: CustomPaint(painter: SteamPainter(time: _time)),
                      ),
                    ),
                  // Equipment
                  Positioned(
                    bottom: 20,
                    child: SizedBox(
                      width: s.width + 60,
                      height: s.height + 20,
                      child: CustomPaint(
                        painter: EquipPainter(
                          shape: eq.shape,
                          liquidColor: liquidColor,
                          fillRatio: eq.fillRatio,
                          isHovered: isHovered,
                          bubbles: isActive && (currentReaction?.bubbles ?? false),
                          precipitate: isActive && (currentReaction?.precipitate ?? false),
                          precipitateColor: currentReaction?.colorChange ?? P.blueBright,
                          isStirring: eq.isStirring,
                          time: _time,
                          hasThermo: eq.hasThermo,
                          hasDropper: eq.hasDropper,
                          temperature: eq.temperature,
                        ),
                      ),
                    ),
                  ),
                ],
              ),
            ),
          ],
        ),
        // Remove button
        Positioned(
          top: 0,
          right: 0,
          child: GestureDetector(
            onTap: () => removeEquipment(eq.id),
            child: Container(
              width: 20, height: 20,
              decoration: BoxDecoration(
                color: P.red,
                shape: BoxShape.circle,
                border: Border.all(color: Colors.white, width: 1.2),
                boxShadow: [BoxShadow(color: Colors.black.withValues(alpha: 0.4), blurRadius: 4)],
              ),
              child: const Icon(Icons.close, size: 11, color: Colors.white),
            ),
          ),
        ),
      ],
    );
  }

// BOTTLE ICON PAINTER
// ═══════════════════════════════════════════════════════════
class BottleIconPainter extends CustomPainter {
  final Color color;
  BottleIconPainter({required this.color});

  @override
  void paint(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;

    // Cap
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(w * 0.32, 0, w * 0.36, h * 0.12),
        const Radius.circular(2),
      ),
      Paint()..color = const Color(0xFF475569),
    );

    // Neck
    canvas.drawRect(
      Rect.fromLTWH(w * 0.4, h * 0.12, w * 0.2, h * 0.1),
      Paint()..color = color.withValues(alpha: 0.7),
    );

    // Shoulder
    final shoulder = Path()
      ..moveTo(w * 0.4, h * 0.22)
      ..quadraticBezierTo(w * 0.1, h * 0.3, w * 0.1, h * 0.4)
      ..lineTo(w * 0.9, h * 0.4)
      ..quadraticBezierTo(w * 0.9, h * 0.3, w * 0.6, h * 0.22)
      ..close();
    canvas.drawPath(
      shoulder,
      Paint()
        ..shader = LinearGradient(
          colors: [color.withValues(alpha: 0.6), color.withValues(alpha: 0.95)],
        ).createShader(Rect.fromLTWH(0, h * 0.22, w, h * 0.18)),
    );

    // Body
    final body = RRect.fromRectAndRadius(
      Rect.fromLTWH(w * 0.1, h * 0.4, w * 0.8, h * 0.58),
      const Radius.circular(4),
    );
    canvas.drawRRect(
      body,
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topLeft, end: Alignment.bottomRight,
          colors: [
            color.withValues(alpha: 0.5),
            color.withValues(alpha: 0.9),
            color,
          ],
        ).createShader(body.outerRect),
    );

    // Border
    canvas.drawRRect(
      body,
      Paint()
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1
        ..color = Colors.black.withValues(alpha: 0.3),
    );

    // Shine
    canvas.drawRect(
      Rect.fromLTWH(w * 0.16, h * 0.45, w * 0.06, h * 0.45),
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topCenter, end: Alignment.bottomCenter,
          colors: [Colors.white.withValues(alpha: 0.8), Colors.white.withValues(alpha: 0.0)],
        ).createShader(Rect.fromLTWH(w * 0.16, h * 0.45, w * 0.06, h * 0.45)),
    );

    // Label
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(w * 0.2, h * 0.62, w * 0.6, h * 0.22),
        const Radius.circular(2),
      ),
      Paint()..color = Colors.white.withValues(alpha: 0.9),
    );
  }

  @override
  bool shouldRepaint(covariant BottleIconPainter old) => old.color != color;
}

// ═══════════════════════════════════════════════════════════
// FLAME PAINTER (below equipment)
// ═══════════════════════════════════════════════════════════
class FlamePainter extends CustomPainter {
  final double progress;
  FlamePainter({required this.progress});

  @override
  void paint(Canvas canvas, Size size) {
    final cx = size.width / 2;
    final flick = sin(progress * 6) * 4;
    final fh = size.height * 0.75 + flick;

    // Outer glow
    canvas.drawCircle(
      Offset(cx, size.height - 5),
      45,
      Paint()
        ..shader = RadialGradient(
          colors: [P.orange.withValues(alpha: 0.35), Colors.transparent],
        ).createShader(Rect.fromCircle(center: Offset(cx, size.height - 5), radius: 45)),
    );

    // Outer flame (orange)
    final outer = Path()
      ..moveTo(cx - 22, size.height)
      ..quadraticBezierTo(cx - 35, size.height - fh * 0.5, cx, size.height - fh)
      ..quadraticBezierTo(cx + 35, size.height - fh * 0.5, cx + 22, size.height)
      ..close();
    canvas.drawPath(
      outer,
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.bottomCenter, end: Alignment.topCenter,
          colors: [
            Colors.deepOrange.withValues(alpha: 0.95),
            Colors.orange.withValues(alpha: 0.85),
            Colors.yellow.withValues(alpha: 0.7),
          ],
        ).createShader(Rect.fromLTWH(cx - 35, size.height - fh, 70, fh)),
    );

    // Inner flame (blue)
    final inner = Path()
      ..moveTo(cx - 10, size.height)
      ..quadraticBezierTo(cx - 15, size.height - fh * 0.35, cx, size.height - fh * 0.7)
      ..quadraticBezierTo(cx + 15, size.height - fh * 0.35, cx + 10, size.height)
      ..close();
    canvas.drawPath(
      inner,
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.bottomCenter, end: Alignment.topCenter,
          colors: [Colors.blue.shade400, Colors.lightBlue.shade200],
        ).createShader(Rect.fromLTWH(cx - 15, size.height - fh * 0.7, 30, fh * 0.7)),
    );
  }

  @override
  bool shouldRepaint(covariant FlamePainter old) => true;
}

// ═══════════════════════════════════════════════════════════
// ICE / FROST PAINTER
// ═══════════════════════════════════════════════════════════
class IcePainter extends CustomPainter {
  final double progress;
  IcePainter({required this.progress});

  @override
  void paint(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;
    final rng = Random(42);

    // Frost particles around
    for (int i = 0; i < 12; i++) {
      final angle = (i / 12) * pi * 2 + progress * 0.3;
      final rx = (w * 0.5 - 10) * 0.8;
      final ry = (h * 0.5 - 15) * 0.8;
      final x = w / 2 + cos(angle) * rx;
      final y = h / 2 + sin(angle) * ry;

      canvas.drawCircle(
        Offset(x, y),
        2 + rng.nextDouble() * 2,
        Paint()..color = Colors.lightBlue.shade100.withValues(alpha: 0.7),
      );
    }

    // Snowflakes at corners
    for (int i = 0; i < 4; i++) {
      final angle = (i / 4) * pi * 2 + progress * 0.2;
      final x = w / 2 + cos(angle) * (w * 0.4);
      final y = h / 2 + sin(angle) * (h * 0.35);

      final p = Paint()
        ..color = Colors.lightBlue.shade200.withValues(alpha: 0.8)
        ..strokeWidth = 1.2;
      canvas.drawLine(Offset(x - 4, y), Offset(x + 4, y), p);
      canvas.drawLine(Offset(x, y - 4), Offset(x, y + 4), p);
    }
  }

  @override
  bool shouldRepaint(covariant IcePainter old) => true;
}

// ═══════════════════════════════════════════════════════════
// STEAM PAINTER
// ═══════════════════════════════════════════════════════════
class SteamPainter extends CustomPainter {
  final double time;
  SteamPainter({required this.time});

  @override
  void paint(Canvas canvas, Size size) {
    final rng = Random(13);
    for (int i = 0; i < 10; i++) {
      final baseX = size.width * 0.5 + (rng.nextDouble() - 0.5) * size.width * 0.5;
      final delay = rng.nextDouble();
      final p = (time * 0.4 + delay) % 1.0;
      final y = size.height * (1 - p);
      final x = baseX + sin(p * pi * 3 + i) * 8;
      final r = 4 + p * 14;
      final opacity = (1 - p) * 0.5;

      canvas.drawCircle(
        Offset(x, y),
        r,
        Paint()
          ..color = Colors.white.withValues(alpha: opacity)
          ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 6),
      );
    }
  }

  @override
  bool shouldRepaint(covariant SteamPainter old) => true;
}

// ═══════════════════════════════════════════════════════════
// EQUIPMENT PAINTER (real shapes)
// ═══════════════════════════════════════════════════════════
class EquipPainter extends CustomPainter {
  final EquipShape shape;
  final Color liquidColor;
  final double fillRatio;
  final bool isHovered;
  final bool bubbles;
  final bool precipitate;
  final Color precipitateColor;
  final bool isStirring;
  final double time;
  final bool hasThermo;
  final bool hasDropper;
  final double temperature;

  EquipPainter({
    required this.shape,
    required this.liquidColor,
    required this.fillRatio,
    required this.isHovered,
    required this.bubbles,
    required this.precipitate,
    required this.precipitateColor,
    required this.isStirring,
    required this.time,
    required this.hasThermo,
    required this.hasDropper,
    required this.temperature,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final ew = shape.size.width;
    final eh = shape.size.height;
    final ox = (size.width - ew) / 2;
    final oy = (size.height - eh) / 2;

    canvas.save();
    canvas.translate(ox, oy);

    final path = _buildPath(ew, eh);

    // Drop shadow
    canvas.drawOval(
      Rect.fromLTWH(ew * 0.15, eh * 0.94, ew * 0.7, eh * 0.07),
      Paint()
        ..color = Colors.black.withValues(alpha: 0.5)
        ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 8),
    );

    // Glass tint (very subtle - transparent)
    canvas.drawPath(
      path,
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topLeft, end: Alignment.bottomRight,
          colors: [
            Colors.white.withValues(alpha: 0.12),
            Colors.white.withValues(alpha: 0.03),
          ],
        ).createShader(Rect.fromLTWH(0, 0, ew, eh)),
    );

    // Liquid + particles
    canvas.save();
    canvas.clipPath(path);

    if (liquidColor != Colors.transparent && fillRatio > 0) {
      _paintLiquid(canvas, ew, eh);
      if (bubbles) _paintBubbles(canvas, ew, eh);
      if (precipitate) _paintPrecipitate(canvas, ew, eh);
      if (isStirring) _paintStirRod(canvas, ew, eh);
    }

    canvas.restore();

    // Glass outline (thicker, clear)
    canvas.drawPath(
      path,
      Paint()
        ..style = PaintingStyle.stroke
        ..strokeWidth = isHovered ? 3 : 2
        ..strokeJoin = StrokeJoin.round
        ..color = isHovered ? P.green : Colors.white.withValues(alpha: 0.85),
    );

    // Hover glow
    if (isHovered) {
      canvas.drawPath(
        path,
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 6
          ..color = P.green.withValues(alpha: 0.45)
          ..maskFilter = const MaskFilter.blur(BlurStyle.outer, 12),
      );
    }

    // Glass shine (left highlight)
    canvas.drawRect(
      Rect.fromLTWH(ew * 0.16, eh * 0.12, ew * 0.04, eh * 0.65),
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topCenter, end: Alignment.bottomCenter,
          colors: [Colors.white.withValues(alpha: 0.7), Colors.white.withValues(alpha: 0.0)],
        ).createShader(Rect.fromLTWH(ew * 0.16, eh * 0.12, ew * 0.04, eh * 0.65)),
    );

    // Right small shine
    canvas.drawRect(
      Rect.fromLTWH(ew * 0.85, eh * 0.3, ew * 0.02, eh * 0.4),
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topCenter, end: Alignment.bottomCenter,
          colors: [Colors.white.withValues(alpha: 0.4), Colors.white.withValues(alpha: 0.0)],
        ).createShader(Rect.fromLTWH(ew * 0.85, eh * 0.3, ew * 0.02, eh * 0.4)),
    );

    // Measurement marks
    _paintMarks(canvas, ew, eh);

    // Thermometer overlay
    if (hasThermo) _paintThermo(canvas, ew, eh);

    // Dropper overlay
    if (hasDropper) _paintDropper(canvas, ew, eh);

    canvas.restore();
  }

  Path _buildPath(double w, double h) {
    final path = Path();
    switch (shape) {
      case EquipShape.beaker:
        path.moveTo(w * 0.08, h * 0.08);
        path.lineTo(w * 0.85, h * 0.08);
        path.lineTo(w * 0.95, h * 0.11);
        path.lineTo(w * 0.87, h * 0.14);
        path.lineTo(w * 0.92, h * 0.14);
        path.lineTo(w * 0.92, h * 0.90);
        path.quadraticBezierTo(w * 0.92, h * 0.96, w * 0.82, h * 0.96);
        path.lineTo(w * 0.18, h * 0.96);
        path.quadraticBezierTo(w * 0.08, h * 0.96, w * 0.08, h * 0.90);
        path.close();
        break;
      case EquipShape.flask:
        path.moveTo(w * 0.38, h * 0.05);
        path.lineTo(w * 0.62, h * 0.05);
        path.lineTo(w * 0.62, h * 0.28);
        path.quadraticBezierTo(w * 0.98, h * 0.55, w * 0.98, h * 0.86);
        path.quadraticBezierTo(w * 0.98, h * 0.96, w * 0.84, h * 0.96);
        path.lineTo(w * 0.16, h * 0.96);
        path.quadraticBezierTo(w * 0.02, h * 0.96, w * 0.02, h * 0.86);
        path.quadraticBezierTo(w * 0.02, h * 0.55, w * 0.38, h * 0.28);
        path.close();
        break;
      case EquipShape.testTube:
        path.moveTo(w * 0.18, h * 0.05);
        path.lineTo(w * 0.82, h * 0.05);
        path.lineTo(w * 0.85, h * 0.08);
        path.lineTo(w * 0.82, h * 0.09);
        path.lineTo(w * 0.82, h * 0.82);
        path.quadraticBezierTo(w * 0.82, h * 0.96, w * 0.5, h * 0.96);
        path.quadraticBezierTo(w * 0.18, h * 0.96, w * 0.18, h * 0.82);
        path.close();
        break;
      case EquipShape.cylinder:
        path.moveTo(w * 0.3, h * 0.05);
        path.lineTo(w * 0.7, h * 0.05);
        path.lineTo(w * 0.73, h * 0.08);
        path.lineTo(w * 0.7, h * 0.09);
        path.lineTo(w * 0.7, h * 0.86);
        path.lineTo(w * 0.92, h * 0.94);
        path.lineTo(w * 0.92, h * 0.96);
        path.lineTo(w * 0.08, h * 0.96);
        path.lineTo(w * 0.08, h * 0.94);
        path.lineTo(w * 0.3, h * 0.86);
        path.close();
        break;
    }
    return path;
  }

  void _paintLiquid(Canvas canvas, double w, double h) {
    final top = h * (1 - 0.85 * fillRatio) - h * 0.04;
    final rect = Rect.fromLTWH(0, top, w, h - top);

    canvas.drawRect(
      rect,
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topCenter, end: Alignment.bottomCenter,
          colors: [
            liquidColor.withValues(alpha: 0.55),
            liquidColor.withValues(alpha: 0.85),
            liquidColor.withValues(alpha: 0.98),
          ],
          stops: const [0.0, 0.5, 1.0],
        ).createShader(rect),
    );

    // Meniscus
    final meniscus = Path()
      ..moveTo(0, top)
      ..quadraticBezierTo(w * 0.5, top + 4, w, top);
    canvas.drawPath(
      meniscus,
      Paint()
        ..style = PaintingStyle.stroke
        ..strokeWidth = 2
        ..color = Colors.white.withValues(alpha: 0.5),
    );
    canvas.drawRect(
      Rect.fromLTWH(0, top, w, 1.5),
      Paint()..color = Colors.white.withValues(alpha: 0.7),
    );
  }

  void _paintBubbles(Canvas canvas, double w, double h) {
    final top = h * (1 - 0.85 * fillRatio) - h * 0.04;
    final rng = Random(11);
    for (int i = 0; i < 14; i++) {
      final x = rng.nextDouble() * w;
      final y = top + rng.nextDouble() * (h - top);
      final r = 1.5 + rng.nextDouble() * 4;

      canvas.drawCircle(Offset(x, y), r, Paint()..color = Colors.white.withValues(alpha: 0.4));
      canvas.drawCircle(
        Offset(x, y), r,
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 0.8
          ..color = Colors.white.withValues(alpha: 0.85),
      );
    }
  }

  void _paintPrecipitate(Canvas canvas, double w, double h) {
    final pHeight = h * 0.22;
    final rect = Rect.fromLTWH(0, h - pHeight, w, pHeight);
    canvas.drawRect(
      rect,
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topCenter, end: Alignment.bottomCenter,
          colors: [
            precipitateColor.withValues(alpha: 0.4),
            precipitateColor.withValues(alpha: 0.95),
          ],
        ).createShader(rect),
    );

    final rng = Random(42);
    for (int i = 0; i < 25; i++) {
      canvas.drawCircle(
        Offset(rng.nextDouble() * w, h - rng.nextDouble() * pHeight),
        1.0 + rng.nextDouble() * 1.8,
        Paint()..color = precipitateColor.withValues(alpha: 0.85),
      );
    }
  }

  void _paintStirRod(Canvas canvas, double w, double h) {
    final angle = time * 4;
    final cx = w * 0.5 + sin(angle) * w * 0.22;
    canvas.drawLine(
      Offset(cx, 0),
      Offset(cx, h * 0.85),
      Paint()
        ..color = Colors.white.withValues(alpha: 0.85)
        ..strokeWidth = 3
        ..strokeCap = StrokeCap.round,
    );
    canvas.drawLine(
      Offset(cx - 1, 0),
      Offset(cx - 1, h * 0.85),
      Paint()
        ..color = Colors.white.withValues(alpha: 0.5)
        ..strokeWidth = 1,
    );
  }

  void _paintMarks(Canvas canvas, double w, double h) {
    final marks = [50, 100, 150, 200, 250];
    for (int i = 0; i < marks.length; i++) {
      final y = h * (0.82 - (i / marks.length) * 0.65);
      final isLong = i.isEven;
      final markW = isLong ? w * 0.16 : w * 0.09;

      canvas.drawLine(
        Offset(w - markW - 3, y),
        Offset(w - 4, y),
        Paint()
          ..color = Colors.white.withValues(alpha: 0.75)
          ..strokeWidth = 1.2,
      );

      final tp = TextPainter(
        text: TextSpan(
          text: '${marks[i]}',
          style: TextStyle(
            fontSize: 7,
            color: Colors.white.withValues(alpha: 0.85),
            fontWeight: FontWeight.bold,
          ),
        ),
        textDirection: TextDirection.ltr,
      )..layout();
      tp.paint(canvas, Offset(w - markW - tp.width - 6, y - 4));
    }
  }

  void _paintThermo(Canvas canvas, double w, double h) {
    final tx = w * 0.35;
    // Tube
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(tx, -h * 0.2, 10, h * 0.8),
        const Radius.circular(5),
      ),
      Paint()..color = Colors.white.withValues(alpha: 0.85),
    );
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(tx, -h * 0.2, 10, h * 0.8),
        const Radius.circular(5),
      ),
      Paint()
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1
        ..color = Colors.grey.shade600,
    );
    // Mercury
    final level = ((temperature - 20) / 130).clamp(0.05, 1.0);
    final mercuryH = h * 0.7 * level;
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(tx + 2, h * 0.6 - mercuryH, 6, mercuryH),
        const Radius.circular(3),
      ),
      Paint()..color = Colors.red.shade700,
    );
    // Bulb
    canvas.drawCircle(Offset(tx + 5, h * 0.62), 9, Paint()..color = Colors.red.shade700);
  }

  void _paintDropper(Canvas canvas, double w, double h) {
    final dx = w * 0.7;
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(dx - 6, -h * 0.15, 12, h * 0.4),
        const Radius.circular(6),
      ),
      Paint()..color = Colors.grey.shade700,
    );
    canvas.drawRect(
      Rect.fromLTWH(dx - 2, h * 0.25, 4, h * 0.25),
      Paint()..color = Colors.white.withValues(alpha: 0.6),
    );
    canvas.drawCircle(Offset(dx, h * 0.5), 2.5, Paint()..color = P.purple);
    // Falling drop
    final dropP = (time * 0.6) % 1.0;
    canvas.drawCircle(
      Offset(dx, h * 0.5 + dropP * h * 0.25),
      2.5,
      Paint()..color = P.purple.withValues(alpha: 1 - dropP),
    );
  }

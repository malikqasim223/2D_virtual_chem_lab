class EquipmentData {
  final EquipmentType type;
  final Map<String, double> chemicalVolumes = {};
  double temperature = 25.0;
  bool isHeating = false;
  bool isStirring = false;
  EquipmentData(this.type);
  double get totalVolume => chemicalVolumes.values.fold(0.0, (a, b) => a + b);
  List<String> get chemicalIds => chemicalVolumes.keys.toList();
  void addChemical(String id, double mL) {
    chemicalVolumes[id] = (chemicalVolumes[id] ?? 0.0) + mL;
    if (chemicalVolumes[id]! > type.maxVolume) chemicalVolumes[id] = type.maxVolume;
  }
  void clear() { chemicalVolumes.clear(); temperature = 25.0; isHeating = false; isStirring = false; }
}

class ToolData {
  final ToolType type;
  bool isActive = false;
  ToolData(this.type);
}

  @override
  void initState() {
    super.initState();
    _ensureEquipCapacity();
    _heatTimer = Timer.periodic(const Duration(milliseconds: 900), (_) => _tickHeating());
  }

  void _ensureEquipCapacity() {
    while (_bubbleCtrl.length < _equipment.length) {
      _bubbleCtrl.add(AnimationController(vsync: this, duration: const Duration(seconds: 3)));
      _pourCtrl.add(AnimationController(vsync: this, duration: const Duration(milliseconds: 800)));
      _shakeCtrl.add(AnimationController(vsync: this, duration: const Duration(milliseconds: 600)));
      _steamCtrl.add(AnimationController(vsync: this, duration: const Duration(seconds: 4))..repeat());
      _sparkCtrl.add(AnimationController(vsync: this, duration: const Duration(seconds: 2)));
      _results.add(null);
      _showSteam.add(false);
      _showSparks.add(false);
      _equipKeys.add(GlobalKey());
    }
  }

  void _ensureToolCapacity() {
    while (_toolCtrl.length < _tools.length) {
      _toolCtrl.add(AnimationController(vsync: this, duration: const Duration(seconds: 2))..repeat());
      _toolKeys.add(GlobalKey());
    }
  }

  @override
  void dispose() {
    _heatTimer?.cancel();
    _dragOverlay?.remove();
    for (final c in _bubbleCtrl) { c.dispose(); }
    for (final c in _pourCtrl) { c.dispose(); }
    for (final c in _shakeCtrl) { c.dispose(); }
    for (final c in _steamCtrl) { c.dispose(); }
    for (final c in _sparkCtrl) { c.dispose(); }
    for (final c in _toolCtrl) { c.dispose(); }
    super.dispose();
  }

  void _addEquipment(EquipmentType t) {
    SystemSound.play(SystemSoundType.click);
    setState(() { _equipment.add(EquipmentData(t)); _ensureEquipCapacity(); });
  }

  void _removeEquipment(int i) {
    SystemSound.play(SystemSoundType.click);
    setState(() {
      if (_equipment.length <= 1) return;
      _equipment.removeAt(i);
      _bubbleCtrl[i].dispose(); _bubbleCtrl.removeAt(i);
      _pourCtrl[i].dispose(); _pourCtrl.removeAt(i);
      _shakeCtrl[i].dispose(); _shakeCtrl.removeAt(i);
      _steamCtrl[i].dispose(); _steamCtrl.removeAt(i);
      _sparkCtrl[i].dispose(); _sparkCtrl.removeAt(i);
      _results.removeAt(i); _showSteam.removeAt(i);
      _showSparks.removeAt(i); _equipKeys.removeAt(i);
    });
  }

  void _addTool(ToolType t) {
    SystemSound.play(SystemSoundType.click);
    setState(() { _tools.add(ToolData(t)); _ensureToolCapacity(); });
  }

  void _removeTool(int i) {
    SystemSound.play(SystemSoundType.click);
    setState(() {
      _tools.removeAt(i);
      _toolCtrl[i].dispose(); _toolCtrl.removeAt(i);
      _toolKeys.removeAt(i);
    });
  }

  void _toggleTool(int i) {
    SystemSound.play(SystemSoundType.click);
    final tool = _tools[i];
    setState(() => tool.isActive = !tool.isActive);
    if (tool.type == ToolType.bunsenBurner) {
      for (var k = 0; k < _equipment.length; k++) {
        _equipment[k].isHeating = tool.isActive;
        _updateTemp(k);
      }
    } else if (tool.type == ToolType.stirrer) {
      for (var k = 0; k < _equipment.length; k++) {
        _equipment[k].isStirring = tool.isActive;
      }
    }
  }

  void _updateTemp(int i) {
    final eq = _equipment[i];
    if (eq.isHeating) {
      eq.temperature = (eq.temperature + 4).clamp(25.0, 150.0);
    } else if (eq.temperature > 25.0) {
      eq.temperature = (eq.temperature - 2).clamp(25.0, 150.0);
    }
    _showSteam[i] = eq.temperature > 60.0 && eq.totalVolume > 0;
  }

  void _tickHeating() {
    if (!mounted) return;
    setState(() {
      for (var i = 0; i < _equipment.length; i++) {
        if (_equipment[i].isHeating || _equipment[i].temperature > 25.0) {
          _updateTemp(i);
        }
      }
    });
  }

  void _startDrag(Chemical c, Offset pos) {
    _draggingChemical = c;
    _dragPos.value = pos;
    _dragOverlay = OverlayEntry(builder: (_) => ValueListenableBuilder<Offset>(
      valueListenable: _dragPos,
      builder: (_, p, __) => Positioned(left: p.dx - 40, top: p.dy - 60,
        child: IgnorePointer(child: Material(color: Colors.transparent, child: _floatingBottle(c)))),
    ));
    Overlay.of(context).insert(_dragOverlay!);
    setState(() {});
  }

  void _updateDrag(Offset pos) {
    if (_dragOverlay == null) return;
    _dragPos.value = pos;
    final idx = _equipAtPoint(pos);
    if (idx != _hoveredIndex) setState(() => _hoveredIndex = idx);
  }

  void _endDrag() {
    _dragOverlay?.remove();
    _dragOverlay = null;
    final idx = _equipAtPoint(_dragPos.value);
    if (idx != null && _draggingChemical != null) _pourInto(idx, _draggingChemical!);
    setState(() { _draggingChemical = null; _hoveredIndex = null; });
  }

  int? _equipAtPoint(Offset pos) {
    for (var i = 0; i < _equipment.length; i++) {
      final ctx = _equipKeys[i].currentContext;
      if (ctx == null) continue;
      final box = ctx.findRenderObject() as RenderBox?;
      if (box == null) continue;
      final local = box.globalToLocal(pos);
      if (local.dx >= -20 && local.dy >= -20 && local.dx <= box.size.width + 20 && local.dy <= box.size.height + 20) return i;
    }
    return null;
  }

  void _pourInto(int i, Chemical c) {
    SystemSound.play(SystemSoundType.click);
    setState(() {
      final eq = _equipment[i];
      eq.addChemical(c.id, _selectedVolume);
      final prev = _results[i];
      final next = ReactionEngine.verify(eq.chemicalIds);
      _results[i] = next;
      _pourCtrl[i].forward(from: 0);
      if (next.matched && (prev == null || !prev.matched)) {
        _shakeCtrl[i].forward(from: 0);
        _sparkCtrl[i].forward(from: 0);
        _showSparks[i] = true;
        SystemSound.play(SystemSoundType.alert);
        Future.delayed(const Duration(milliseconds: 1800), () {
          if (mounted) setState(() => _showSparks[i] = false);
        });
      }
      if (next.reaction?.bubbles == true) { _bubbleCtrl[i].repeat(); }
      else { _bubbleCtrl[i].stop(); }
    });
  }

  void _reset() {
    SystemSound.play(SystemSoundType.click);
    setState(() {
      for (var i = 0; i < _equipment.length; i++) {
        _equipment[i].clear();
        _results[i] = null;
        _bubbleCtrl[i].stop(); _bubbleCtrl[i].reset();
        _pourCtrl[i].reset(); _shakeCtrl[i].reset();
        _showSteam[i] = false; _showSparks[i] = false;
      }
      _hoveredIndex = null;
    });
  }

  Color _liquidColor(int i) {
    final r = _results[i];
    if (r?.matched == true && r?.reaction?.colorChange != null) return r!.reaction!.colorChange!;
    final eq = _equipment[i];
    if (eq.chemicalIds.isEmpty) return Colors.transparent;
    return _chemicals.firstWhere((c) => c.id == eq.chemicalIds.last).color;
  }

  Widget _equipmentWidget(int i) {
    final eq = _equipment[i];
    final hover = _hoveredIndex == i;
    final color = _liquidColor(i);
    final total = eq.totalVolume;
    final maxV = eq.type.maxVolume;
    final fill = (total / maxV).clamp(0.0, 1.0);
    final size = eq.type.size;

    return Stack(
      clipBehavior: Clip.none,
      children: [
        Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            _equipLabel(eq, total, maxV, hover),
            const SizedBox(height: 8),
            AnimatedBuilder(
              animation: _shakeCtrl[i],
              builder: (_, child) {
                final p = _shakeCtrl[i].value;
                final off = p > 0 ? sin(p * pi * 8) * (1 - p) * 8 : 0.0;
                return Transform.translate(offset: Offset(off, 0), child: child);
              },
              child: SizedBox(
                width: size.width + 20,
                height: size.height + 60,
                child: Stack(
                  alignment: Alignment.topCenter,
                  children: [
                    if (_showSteam[i])
                      Positioned(
                        top: 0, left: 10, right: 10, height: 60,
                        child: AnimatedBuilder(
                          animation: _steamCtrl[i],
                          builder: (_, __) => CustomPaint(
                            painter: SteamPainter(progress: _steamCtrl[i].value),
                          ),
                        ),
                      ),
                    Positioned(
                      bottom: 0,
                      child: AnimatedBuilder(
                        animation: Listenable.merge([_pourCtrl[i], _bubbleCtrl[i]]),
                        builder: (_, __) {
                          return Container(
                            key: _equipKeys[i],
                            width: size.width,
                            height: size.height,
                            child: CustomPaint(
                              painter: EquipmentPainter(
                                shape: eq.type.shape,
                                liquidColor: color,
                                fillRatio: fill,
                                isHovering: hover,
                                showBubbles: _results[i]?.reaction?.bubbles == true,
                                showPrecipitate: _results[i]?.reaction?.precipitate == true,
                                precipitateColor: _results[i]?.reaction?.colorChange ?? const Color(0xFF60A5FA),
                                pourProgress: _pourCtrl[i].value,
                                bubbleProgress: _bubbleCtrl[i].value,
                                isStirring: eq.isStirring,
                                stirProgress: _toolCtrl.isNotEmpty ? _toolCtrl.first.value : 0.0,
                              ),
                            ),
                          );
                        },
                      ),
                    ),
                    if (_showSparks[i])
                      Positioned.fill(
                        child: IgnorePointer(
                          child: AnimatedBuilder(
                            animation: _sparkCtrl[i],
                            builder: (_, __) => CustomPaint(
                              painter: SparksPainter(progress: _sparkCtrl[i].value, color: K.amber),
                            ),
                          ),
                        ),
                      ),
                  ],
                ),
              ),
            ),
          ],
        ),
        Positioned(
          top: 40, right: -6,
          child: GestureDetector(
            onTap: () => _removeEquipment(i),
            child: Container(
              width: 18, height: 18,
              decoration: BoxDecoration(
                color: K.red, shape: BoxShape.circle,
                border: Border.all(color: Colors.white, width: 1.2),
              ),
              child: const Icon(Icons.close, size: 9, color: Colors.white),
            ),
          ),
        ),
      ],
    );
  }

  Widget _reactionContent(int idx) {
    final r = _results[idx]!;
    if (!r.matched) {
      return Container(
        padding: const EdgeInsets.all(10),
        decoration: BoxDecoration(
          color: K.bg,
          borderRadius: BorderRadius.circular(6),
          border: Border.all(color: K.red.withValues(alpha: 0.3)),

            Icon(Icons.error_outline, size: 14, color: K.red),
            SizedBox(width: 8),
            Expanded(child: Text('No supported reaction', style: TextStyle(color: K.red, fontSize: 10))),
          ],
        ),
      );
    }

    final rxn = r.reaction!;
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        _infoRow('Name', rxn.name),
        _infoRow('Type', rxn.type),
        const SizedBox(height: 8),
        Container(
          width: double.infinity,

          ),
          child: Text(rxn.equation,
            style: const TextStyle(color: K.cyan, fontSize: 12, fontWeight: FontWeight.bold),
            textAlign: TextAlign.center),
        ),
        const SizedBox(height: 8),
        Text(rxn.description,

class EquipmentPainter extends CustomPainter {
  final EqShape shape;
  final Color liquidColor;
  final double fillRatio;
  final bool isHovering;
  final bool showBubbles;
  final bool showPrecipitate;
  final Color precipitateColor;
  final double pourProgress;
  final double bubbleProgress;
  final bool isStirring;
  final double stirProgress;

  EquipmentPainter({
    required this.shape,
    required this.liquidColor,
    required this.fillRatio,
    required this.isHovering,
    required this.showBubbles,
    required this.showPrecipitate,
    required this.precipitateColor,
    required this.pourProgress,
    required this.bubbleProgress,
    required this.isStirring,
    required this.stirProgress,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final path = _buildPath(size);

    canvas.drawPath(
      path,
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topLeft,
          end: Alignment.bottomRight,
          colors: [
            Colors.white.withValues(alpha: 0.35),
            Colors.white.withValues(alpha: 0.08),
          ],
        ).createShader(Offset.zero & size),
    );

    canvas.save();
    canvas.clipPath(path);

    if (fillRatio > 0) {
      _liquid(canvas, size);
      if (showPrecipitate) _precipitate(canvas, size);
      if (showBubbles) _bubbles(canvas, size);
    }
    if (pourProgress > 0 && pourProgress < 1) _pourStream(canvas, size);
    if (isStirring) _stirRod(canvas, size);

    canvas.restore();

    canvas.drawPath(
      path,
      Paint()
        ..style = PaintingStyle.stroke
        ..strokeWidth = isHovering ? 2.5 : 1.8
        ..strokeJoin = StrokeJoin.round
        ..color = isHovering ? K.green : Colors.white.withValues(alpha: 0.75),
    );

    if (isHovering) {
      canvas.drawPath(
        path,
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 6
          ..color = K.green.withValues(alpha: 0.4)
          ..maskFilter = const MaskFilter.blur(BlurStyle.outer, 12),
      );
    }

    _shine(canvas, size);
    _marks(canvas, size);
  }

  Path _buildPath(Size size) {
    final w = size.width;
    final h = size.height;
    final path = Path();

    switch (shape) {
      case EqShape.beaker:
        path.moveTo(w * 0.08, h * 0.06);
        path.lineTo(w * 0.86, h * 0.06);
        path.lineTo(w * 0.94, h * 0.09);
        path.lineTo(w * 0.88, h * 0.12);
        path.lineTo(w * 0.92, h * 0.12);
        path.lineTo(w * 0.92, h * 0.9);
        path.quadraticBezierTo(w * 0.92, h * 0.96, w * 0.82, h * 0.96);
        path.lineTo(w * 0.18, h * 0.96);
        path.quadraticBezierTo(w * 0.08, h * 0.96, w * 0.08, h * 0.9);
        path.close();
        break;
      case EqShape.flask:
        path.moveTo(w * 0.38, h * 0.04);
        path.lineTo(w * 0.62, h * 0.04);
        path.lineTo(w * 0.62, h * 0.24);
        path.quadraticBezierTo(w * 0.98, h * 0.55, w * 0.98, h * 0.84);
        path.quadraticBezierTo(w * 0.98, h * 0.96, w * 0.82, h * 0.96);
        path.lineTo(w * 0.18, h * 0.96);
        path.quadraticBezierTo(w * 0.02, h * 0.96, w * 0.02, h * 0.84);
        path.quadraticBezierTo(w * 0.02, h * 0.55, w * 0.38, h * 0.24);
        path.close();
        break;
      case EqShape.testTube:
        path.moveTo(w * 0.18, h * 0.04);
        path.lineTo(w * 0.82, h * 0.04);
        path.lineTo(w * 0.86, h * 0.07);
        path.lineTo(w * 0.82, h * 0.08);
        path.lineTo(w * 0.82, h * 0.82);
        path.quadraticBezierTo(w * 0.82, h * 0.96, w * 0.5, h * 0.96);
        path.quadraticBezierTo(w * 0.18, h * 0.96, w * 0.18, h * 0.82);
        path.lineTo(w * 0.18, h * 0.08);
        path.close();
        break;
      case EqShape.cylinder:
        path.moveTo(w * 0.3, h * 0.04);
        path.lineTo(w * 0.7, h * 0.04);
        path.lineTo(w * 0.74, h * 0.07);
        path.lineTo(w * 0.7, h * 0.08);
        path.lineTo(w * 0.7, h * 0.85);
        path.lineTo(w * 0.92, h * 0.93);
        path.lineTo(w * 0.92, h * 0.96);
        path.lineTo(w * 0.08, h * 0.96);
        path.lineTo(w * 0.08, h * 0.93);
        path.lineTo(w * 0.3, h * 0.85);
        path.lineTo(w * 0.3, h * 0.08);
        path.close();
        break;
    }
    return path;
  }

  void _liquid(Canvas canvas, Size size) {
    final top = size.height * (1 - 0.9 * fillRatio) - size.height * 0.02;
    final rect = Rect.fromLTWH(0, top, size.width, size.height - top);

    canvas.drawRect(
      rect,
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topCenter,
          end: Alignment.bottomCenter,
          colors: [
            liquidColor.withValues(alpha: 0.6),
            liquidColor.withValues(alpha: 0.95),
          ],
        ).createShader(rect),
    );

    canvas.drawRect(
      Rect.fromLTWH(0, top, size.width, 2.5),
      Paint()..color = Colors.white.withValues(alpha: 0.7),
    );
  }

  void _precipitate(Canvas canvas, Size size) {
    final pHeight = size.height * 0.22;
    final rect = Rect.fromLTWH(0, size.height - pHeight, size.width, pHeight);
    canvas.drawRect(
      rect,
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topCenter,
          end: Alignment.bottomCenter,
          colors: [
            precipitateColor.withValues(alpha: 0.4),
            precipitateColor.withValues(alpha: 0.95),
          ],
        ).createShader(rect),
    );
  }

  void _bubbles(Canvas canvas, Size size) {
    final top = size.height * (1 - 0.9 * fillRatio) - size.height * 0.02;
    final rng = Random(11);
    for (int i = 0; i < 14; i++) {
      final x = 8 + rng.nextDouble() * (size.width - 16);
      final delay = rng.nextDouble();
      final p = (bubbleProgress + delay) % 1.0;
      final y = size.height - (size.height - top) * p - size.height * 0.03;
      final r = 1 + rng.nextDouble() * 3.5;
      if (y < top - 5) continue;
      canvas.drawCircle(
        Offset(x, y),
        r,
        Paint()..color = Colors.white.withValues(alpha: (1 - p) * 0.85),
      );
    }
  }

  void _pourStream(Canvas canvas, Size size) {
    final op = 1 - pourProgress;
    final x = size.width * 0.5;
    canvas.drawLine(
      Offset(x, 0),
      Offset(x, size.height * pourProgress),
      Paint()
        ..color = liquidColor.withValues(alpha: op)
        ..strokeWidth = 3.5
        ..strokeCap = StrokeCap.round,
    );
  }

  void _stirRod(Canvas canvas, Size size) {
    final angle = stirProgress * pi * 4;
    final cx = size.width * 0.5 + sin(angle) * size.width * 0.2;
    canvas.drawLine(
      Offset(cx, 0),
      Offset(cx, size.height * 0.8),
      Paint()
        ..color = Colors.white.withValues(alpha: 0.8)
        ..strokeWidth = 2.5,
    );
  }

  void _shine(Canvas canvas, Size size) {
    canvas.drawRect(
      Rect.fromLTWH(size.width * 0.16, size.height * 0.1, size.width * 0.05, size.height * 0.7),
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topCenter,
          end: Alignment.bottomCenter,
          colors: [
            Colors.white.withValues(alpha: 0.85),
            Colors.white.withValues(alpha: 0.0),
          ],
        ).createShader(Rect.fromLTWH(size.width * 0.16, size.height * 0.1, size.width * 0.05, size.height * 0.7)),
    );
  }

  void _marks(Canvas canvas, Size size) {
    final marks = switch (shape) {
      EqShape.beaker => [50, 100, 150, 200, 250],
      EqShape.flask => [50, 100, 150, 200, 250],
      EqShape.testTube => [10, 20, 30, 40, 50],
      EqShape.cylinder => [20, 40, 60, 80, 100],
    };

    for (int i = 0; i < marks.length; i++) {
      final y = size.height * (0.82 - (i / marks.length) * 0.68);
      final isLong = i.isEven;
      final w = isLong ? size.width * 0.18 : size.width * 0.1;

      canvas.drawLine(
        Offset(size.width - w - 3, y),
        Offset(size.width - 4, y),
        Paint()
          ..color = Colors.white.withValues(alpha: 0.7)
          ..strokeWidth = 1.2,
      );

      final tp = TextPainter(
        text: TextSpan(
          text: '${marks[i]}',
          style: TextStyle(
            fontSize: 7,
            color: Colors.white.withValues(alpha: 0.85),
            fontWeight: FontWeight.bold,
          ),
        ),
        textDirection: TextDirection.ltr,
      )..layout();
      tp.paint(canvas, Offset(size.width - w - tp.width - 6, y - 4));
    }
  }

  @override
  bool shouldRepaint(covariant EquipmentPainter old) => true;
}

class ToolPainter extends CustomPainter {
  final ToolShape shape;
  final bool isActive;
  final double progress;

  ToolPainter({
    required this.shape,
    required this.isActive,
    required this.progress,
  });

  @override
  void paint(Canvas canvas, Size size) {
    switch (shape) {
      case ToolShape.bunsenBurner: _bunsen(canvas, size); break;
      case ToolShape.thermometer: _thermo(canvas, size); break;
      case ToolShape.stirrer: _stirrer(canvas, size); break;
      case ToolShape.balance: _balance(canvas, size); break;
      case ToolShape.dropper: _dropper(canvas, size); break;
      case ToolShape.testTubeRack: _rack(canvas, size); break;
      case ToolShape.filtration: _funnel(canvas, size); break;
    }
  }

  void _bunsen(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;

    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(0, h * 0.85, w, h * 0.15),
        const Radius.circular(3),
      ),
      Paint()..color = const Color(0xFF475569),
    );

    canvas.drawRect(
      Rect.fromLTWH(w * 0.36, h * 0.4, w * 0.28, h * 0.45),
      Paint()..color = const Color(0xFF94A3B8),
    );

    if (isActive) {
      final fh = h * 0.35 + sin(progress * pi * 6) * 3;
      final outer = Path()
        ..moveTo(w * 0.4, h * 0.4)
        ..quadraticBezierTo(w * 0.35, h * 0.4 - fh * 0.5, w * 0.5, h * 0.4 - fh)
        ..quadraticBezierTo(w * 0.65, h * 0.4 - fh * 0.5, w * 0.6, h * 0.4)
        ..close();
      canvas.drawPath(outer, Paint()..color = Colors.orange.withValues(alpha: 0.9));
    }
  }

  void _thermo(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;

    canvas.drawCircle(Offset(w * 0.5, h * 0.92), w * 0.35,
      Paint()..color = Colors.red.shade700);

    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(w * 0.4, h * 0.08, w * 0.2, h * 0.84),
        const Radius.circular(3),
      ),
      Paint()..color = Colors.white.withValues(alpha: 0.9),
    );

    final mh = h * 0.6 * progress;
    canvas.drawRect(
      Rect.fromLTWH(w * 0.44, h * 0.92 - mh, w * 0.12, mh),
      Paint()..color = Colors.red.shade700,
    );
  }

  void _stirrer(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;

    canvas.drawLine(
      Offset(w * 0.5, 0),
      Offset(w * 0.5, h * 0.85),
      Paint()..color = Colors.white.withValues(alpha: 0.9)..strokeWidth = 4,
    );

    if (isActive) {
      canvas.save();
      canvas.translate(w * 0.5, h * 0.1);
      canvas.rotate(progress * pi * 6);
      canvas.drawCircle(
        Offset.zero,
        w * 0.2,
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 2.5
          ..color = K.cyan,
      );
      canvas.restore();
    }
  }

  void _balance(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;

    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(0, h * 0.2, w, h * 0.8),
        const Radius.circular(8),
      ),
      Paint()..color = const Color(0xFFCBD5E1),
    );

    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(w * 0.15, h * 0.38, w * 0.7, h * 0.4),
        const Radius.circular(4),
      ),
      Paint()..color = const Color(0xFF064E3B),
    );

    final tp = TextPainter(
      text: TextSpan(
        text: '${(progress * 100).toInt()} g',
        style: const TextStyle(color: Color(0xFF6EE7B7), fontSize: 16, fontWeight: FontWeight.bold),
      ),
      textDirection: TextDirection.ltr,
    )..layout();
    tp.paint(canvas, Offset(w * 0.5 - tp.width / 2, h * 0.5 - tp.height / 2));
  }

  void _dropper(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;

    canvas.drawCircle(Offset(w * 0.5, h * 0.18), w * 0.35,
      Paint()..color = Colors.grey.shade400);

    canvas.drawRect(
      Rect.fromLTWH(w * 0.4, h * 0.3, w * 0.2, h * 0.55),
      Paint()..color = Colors.white.withValues(alpha: 0.85),
    );

    canvas.drawRect(
      Rect.fromLTWH(w * 0.42, h * 0.55, w * 0.16, h * 0.28),
      Paint()..color = K.cyan.withValues(alpha: 0.6),
    );
  }

  void _rack(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;

    canvas.drawRect(Rect.fromLTWH(0, h * 0.75, w, h * 0.25),
      Paint()..color = const Color(0xFF78350F));
    canvas.drawRect(Rect.fromLTWH(0, h * 0.15, w, h * 0.08),
      Paint()..color = const Color(0xFF92400E));

    final colors = [
      const Color(0xFF7DD3FC),
      const Color(0xFFFBCFE8),
      const Color(0xFFBBF7D0),
      const Color(0xFFFDE68A),
      const Color(0xFFDDD6FE),
      const Color(0xFFFED7AA),
    ];

    for (int i = 0; i < 6; i++) {
      final tx = w * 0.08 + i * w * 0.145;
      canvas.drawRect(Rect.fromLTWH(tx, h * 0.18, w * 0.08, h * 0.6),
        Paint()..color = Colors.white.withValues(alpha: 0.5));
      canvas.drawRect(Rect.fromLTWH(tx, h * 0.5, w * 0.08, h * 0.28),
        Paint()..color = colors[i]);
    }
  }

  void _funnel(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;

    final cone = Path()
      ..moveTo(w * 0.05, h * 0.15)
      ..lineTo(w * 0.95, h * 0.15)
      ..lineTo(w * 0.55, h * 0.6)
      ..lineTo(w * 0.45, h * 0.6)
      ..close();
    canvas.drawPath(cone, Paint()..color = Colors.white.withValues(alpha: 0.6));

    canvas.drawRect(
      Rect.fromLTWH(w * 0.46, h * 0.6, w * 0.08, h * 0.3),
      Paint()..color = Colors.white.withValues(alpha: 0.6),
    );
  }

  @override
  bool shouldRepaint(covariant ToolPainter old) => true;
}

class SteamPainter extends CustomPainter {
  final double progress;
  SteamPainter({required this.progress});

  @override
  void paint(Canvas canvas, Size size) {
    final rng = Random(13);
    for (int i = 0; i < 8; i++) {
      final baseX = 20 + rng.nextDouble() * (size.width - 40);
      final delay = rng.nextDouble();
      final p = (progress + delay) % 1.0;
      final y = size.height * (1 - p);
      final x = baseX + sin(p * pi * 3 + i) * 10;
      final r = 4 + p * 12;

      canvas.drawCircle(
        Offset(x, y),
        r,
        Paint()
          ..color = Colors.white.withValues(alpha: (1 - p) * 0.4)
          ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 6),
      );
    }
  }

  @override
  bool shouldRepaint(covariant SteamPainter old) => old.progress != progress;
}

class SparksPainter extends CustomPainter {
  final double progress;
  final Color color;
  SparksPainter({required this.progress, required this.color});

  @override
  void paint(Canvas canvas, Size size) {
    final rng = Random(21);
    final center = Offset(size.width / 2, size.height / 2);

    for (int i = 0; i < 20; i++) {
      final angle = rng.nextDouble() * pi * 2;
      final speed = 40 + rng.nextDouble() * 60;
      final dist = progress * speed;
      final x = center.dx + cos(angle) * dist;
      final y = center.dy + sin(angle) * dist - progress * progress * 20;

      canvas.drawCircle(
        Offset(x, y),
        2 + rng.nextDouble() * 3,
        Paint()..color = color.withValues(alpha: (1 - progress).clamp(0.0, 1.0)),
      );
    }
  }

  @override
  bool shouldRepaint(covariant SparksPainter old) => old.progress != progress;
}
